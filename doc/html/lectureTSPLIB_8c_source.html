63 fin_lecture_name = true;
64 }
65
66 //Lecture de la dimension
67 else if (strncmp(ligne, "DIMENSION", 9) == 0 && fin_lecture_name) {

69 fin_lecture_dimension = true;

101 }
102 // Lecture des coordonnées
103 else if (!fin_lecture_point && fin_lecture_section) {
104 int id;

114 }
115 //Cas erreur lecture de quelque chose qui n'est pas une coordonnées
116 else {

118 free_graphe(g);
119 fclose(f);

121 }
122 }

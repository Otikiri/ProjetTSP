31/*
32 * Computes the next lexicographical permutation of the specified array of integers in place,
33 * returning a Boolean to indicate whether a next permutation existed.
34 * (Returns false when the argument is already the last possible permutation.)
35 */
36bool next_permutation(int array[], size_t length) {
37 // Find non-increasing suffix
38 if (length == 0)
39 return false;
40 size_t i = length - 1;
41 while (i > 0 && array[i - 1] >= array[i])
42 i--;
43 if (i == 0)
44 return false;
45
46 // Find successor to pivot
47 size_t j = length - 1;
48 while (array[j] <= array[i - 1])
49 j--;
50 int temp = array[i - 1];
51 array[i - 1] = array[j];
52 array[j] = temp;
53
54 // Reverse suffix
55 j = length - 1;
56 while (i < j) {
57 temp = array[i];
58 array[i] = array[j];
59 array[j] = temp;
60 i++;
61 j--;
62 }
63 return true;
64}
